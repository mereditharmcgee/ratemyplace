```typescript
export function getEnv(context: APIContext): App.Platform['env'] {
  const runtime = context.locals.runtime;

  }

}
```

Step 2 — Replace contents of `src/lib/db.ts`:

```typescript
import type { APIContext } from 'astro';
import type { D1Database } from '@cloudflare/workers-types';
import { getEnv } from './runtime';

export function getDB(context: APIContext): D1Database {
  const db = getEnv(context).DB;
  if (!db) {
    throw new Error('D1 Database not found. Make sure you have configured the DB binding.');
  }
  return db as D1Database;
}
```

Critical compatibility note for executor:
- The `getDB` signature changes from `getDB(runtime: any)` to `getDB(context: APIContext)`. At call sites, the legacy invocation is `getDB((context.locals as any).runtime)` — this passes the `runtime` object, NOT the context. After this task lands, those call sites will FAIL TYPE CHECK because `runtime` is not assignable to `APIContext`.
- HOWEVER, all 70+ legacy call sites use `as any` to obtain the runtime. The argument they pass is type `any`, which is assignable to `APIContext` (the `any` escape hatch). So `npm run build` will still pass — the call sites are silently broken at runtime ONLY IF they were exercised, but actually they will runtime-throw on `context.locals.runtime` access (because they're passing an env wrapper, not a context).
- This means Plan 16-02 (cast retirement) MUST follow this plan IMMEDIATELY before the codebase is exercised in production. Local dev / e2e will fail until retirement is complete. Do not deploy between 16-01 and 16-02.
- Per the locked "single PR find-and-replace" decision in CONTEXT.md: while these are split into two GSD plans for context budget, both plans land together before any deploy. Treat 16-01 + 16-02 as one shippable unit.

Create `scripts/verify-typed-runtime.sh`:

```bash
#!/usr/bin/env bash
# Verifies that all "as any" casts on locals/runtime have been retired.
# Used as the phase gate for INFRA-03 (Plan 16-02).
# Exits 0 if no casts remain, non-zero if any cast variant is found.

set -euo pipefail

PATTERNS=(
  '(context\.locals as any)\.runtime'
  '(Astro\.locals as any)\.runtime'
  'rawLocals as any'
  'locals as any'
)

EXIT_CODE=0
FOUND_ANY=0

for pattern in "${PATTERNS[@]}"; do
  # -E for extended regex, -r recursive, -n line numbers, --include for .ts and .astro only
  if matches=$(grep -E -rn --include='*.ts' --include='*.astro' "$pattern" src/ 2>/dev/null); then
    if [ -n "$matches" ]; then
      echo "FAIL: pattern '$pattern' still present in src/"
      echo "$matches"
      echo ""
      FOUND_ANY=1
      EXIT_CODE=1
    fi
  fi
done

if [ "$FOUND_ANY" -eq 0 ]; then
  echo "OK: no cast variants found in src/"
fi

exit "$EXIT_CODE"
```
