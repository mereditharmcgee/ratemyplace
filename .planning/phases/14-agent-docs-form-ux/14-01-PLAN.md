1. Create migration `migrations/0023_saved_buildings.sql`:
   ```sql
   CREATE TABLE IF NOT EXISTS saved_buildings (
     id INTEGER PRIMARY KEY AUTOINCREMENT,
     user_id TEXT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
     building_id TEXT NOT NULL REFERENCES buildings(id) ON DELETE CASCADE,
     created_at INTEGER DEFAULT (unixepoch()),
     UNIQUE(user_id, building_id)
   );
   CREATE INDEX idx_saved_buildings_user ON saved_buildings(user_id);
   ```

2. Add to `src/lib/api-types.ts` (at end, new section):
   ```typescript
   // Saved Buildings Types
   export interface SavedBuilding {
     id: number;
     building_id: string;
     building_address: string;
     building_slug: string;
     neighborhood: string | null;
     city: string;
     state: string;
     review_count: number;

     saved_at: number;

   export interface SavedBuildingsResponse {
   ```
